# Define the reusable function to process and validate the dataset.
def prepare_data(source):
    # Make a separate copy of the original dataset to preserve the original data.
    df = source.copy()
    # Validate whether the dataset is empty.
    if df.empty:
        raise ValueError("The dataset is empty.")
    # Validate whether the churn column is present in the dataset.
    if "churn" not in df.columns:
        raise ValueError("Required column 'churn' is missing (expected 0/1).")
    # Convert the churn values to numeric values and replace any invalid value by a missing value.
    churn = pd.to_numeric(df["churn"], errors="coerce")
    # Check whether the churn values are valid and belong to the expected binary classes 0 and 1.
    if not churn.isin([0, 1]).all():
        # Raise an error if there are any invalid or missing churn values; the values should be mapped to 0 and 1.
        raise ValueError(
            f"'churn' has {int((~churn.isin([0, 1])).sum())} "
            "missing/invalid values. Map them to 0 or 1 first."
        )
    # Convert the churn values to the integer format.
    df["churn"] = churn.astype(int)
    # Map the binary churn values to customer status labels.
    df["churn_label"] = df["churn"].map({0: "Retained", 1: "Churned"})
    # Iterate over the dashboard features and detect numerical features.
    for col in LABELS:
        # Validate whether each of the chosen features is present in the dataset.
        if col in df:
            # Convert the chosen features to numeric values and replace any infinite value by a missing value.
            df[col] = pd.to_numeric(df[col], errors="coerce").replace(
                [np.inf, -np.inf], np.nan
            )
    # Return the cleaned dataset.
    return df

# Defines a reusable function to compute the 95% Wilson confidence interval of a proportion and returns the lower and upper limits in percent.
def wilson_ci(k, n, z=1.96):
    # Converts the observed numbers and the total number of samples to NumPy floating-point arrays.
    k, n = np.asarray(k, float), np.asarray(n, float)
    # Computes the observed proportion by taking the ratio of positive observations to total number of samples.
    p = k / n
    # Computes the denominator of the Wilson confidence interval formula.
    denom = 1 + z ** 2 / n
    # Computes the center of the Wilson confidence interval.
    centre = (p + z ** 2 / (2 * n)) / denom
    # Computes the margin of error of the Wilson confidence interval.
    half = z * np.sqrt(
        p * (1 - p) / n + z ** 2 / (4 * n ** 2)
    ) / denom
    # Returns the lower and upper confidence limits in percent, ensuring the values are within the range of 0 to 100.
    return (
        100 * np.clip(centre - half, 0, 1),
        100 * np.clip(centre + half, 0, 1)
    )

# Creates a function to configure axes and consistent look on the charts.
def style_axes(ax, title, grid_axis="y"):
    # Sets the background color of the chart area using the card color.
    ax.set_facecolor(C_CARD)
    # Sets the title of the chart, including padding, text color and alignment to the left.
    ax.set_title(title, pad=16, color=C_DARK, loc="left")
    # Iterates over top and right axis borders.
    for side in ("top", "right"):
        # Hides top and right axis borders from the chart area.
        ax.spines[side].set_visible(False)
    # Iterates over left and bottom axis borders.
    for side in ("left", "bottom"):
        # Sets the color of left and bottom axis borders.
        ax.spines[side].set_color(C_BORDER)
        # Sets the thickness of left and bottom axis borders.
        ax.spines[side].set_linewidth(1.8)
    # Sets the color of axis ticks.
    ax.tick_params(colors=C_BORDER)
    # Checks whether grid lines should be shown for the chart.
    if grid_axis:
        # Adds grid lines on the specified axis using the defined color and width.
        ax.grid(axis=grid_axis, color=C_GRID, linewidth=1, zorder=0)
    # Adds grid lines behind the chart items.
    ax.set_axisbelow(True)

# Creates a function to show a message when there are no usable data to display a chart.
def no_data(ax, title, msg="No usable data available"):
    # Shows the standard chart styling without grid lines.
    style_axes(ax, title, grid_axis=None)
    # Hides x-axis tick marks when there is no data.
    ax.set_xticks([])
    # Hides y-axis tick marks when there is no data.
    ax.set_yticks([])
    # Shows a centered message inside the chart area saying that there is no usable data.
    ax.text(
        # Specifies the text alignment, font size, bold type and text color.
        0.5, 0.5, msg,
        transform=ax.transAxes,
        ha="center", va="center",
        fontsize=15,
        fontweight="bold",
        color="#757575"
    )

# Defines function to plot churn rate bar charts with confidence intervals, reference line across all, percentages and customer counts.
def rate_bars(ax, groups, rates, counts, overall, title, xlabel, cmap,
              rotate=0, tick_size=17):
    # Explains objective and main elements of churn rate bar chart.
    """Churn-rate bars: own gradient, 95% CI whiskers, overall reference line,
    value pills above the whiskers, n= under each tick label."""
    # Converts the churn rate values to NumPy floating point arrays.
    rates = np.asarray(rates, dtype=float)
    # Converts the customer count values to NumPy floating point arrays.
    counts = np.asarray(counts, dtype=float)
    # Computes the lower and upper ends of the 95% Wilson confidence interval from the churn rates and customer counts.
    lo, hi = wilson_ci(rates / 100 * counts, counts)
    # Computes the difference between the maximum and minimum churn rate to set the colour scale gradient.
    span = max(rates.max() - rates.min(), 1e-9)
    # Sets gradient colours for the bars based on their respective churn rate.
    colors = [
        plt.get_cmap(cmap)(0.30 + 0.65 * (r - rates.min()) / span)
        for r in rates
    ]
    # Computes numerical x positions for the bars.
    x = np.arange(len(rates))
    # Plots the churn rate bar chart with computed x positions, churn rates and colours.
    bars = ax.bar(
        x, rates, color=colors, edgecolor=C_BORDER,
        linewidth=1.8, width=0.66, zorder=3
    )
    # Draws error bars for 95% confidence interval around the churn rates.
    ax.errorbar(
        x, rates, yerr=[rates - lo, hi - rates], fmt="none",
        # Modifies the properties of error bars, such as colour, line width, cap size and legend name.
        ecolor=C_DARK, elinewidth=2.2, capsize=7, capthick=2.2,
        zorder=5, label="95% CI"
    )
    # Plot a reference horizontal dashed line that represents the overall customer churn rate.
    ax.axhline(
        overall, color=C_VIOLET, linestyle="--", linewidth=2.4,
        # Set the parameters for the reference line color, style, line width and legend label.
        zorder=2, label=f"Overall {overall:.1f}%"
    )
    # Find the upper plotting limit based on the highest confidence interval and the overall churn rate.
    top = max(hi.max(), overall)
    # Loop through the data to find the horizontal and vertical coordinates for each percentage label.
    for b_, r, h in zip(bars, rates, hi):
        # Show the churn rate on top of its confidence interval whisker.
        # Format the percentage labels to show only one decimal digit, with alignment, font size, bold format and color settings.
        # Put the white rounded background box behind each label.
        ax.text(
            b_.get_x() + b_.get_width() / 2, h + top * 0.03,
            f"{r:.1f}%", ha="center", va="bottom", fontsize=17,
            fontweight="bold", color=C_DARK, zorder=6,
            bbox=dict(
                boxstyle="round,pad=0.22", fc="white",
                ec=C_GRID, lw=1.2, alpha=0.96
            )
        )
    # Set the x-axis positions corresponding to the different customer groups.
    ax.set_xticks(x)
    # Form the x-axis labels with the customer group names and the number of customers per group.
    ax.set_xticklabels(
        [f"{g}\nn={int(n):,}" for g, n in zip(groups, counts)],
        # Set the rotation angle, horizontal alignment, font size and bold format.
        rotation=rotate, ha="right" if rotate else "center",
        fontsize=tick_size, fontweight="bold"
    )
    # Set the y-axis limits from zero up to the upper plotting limit to leave enough space for labels.
    ax.set_ylim(0, top * 1.55)
    # Limit the y-axis tick values to only the values in the range from 0 to 100.
    ax.set_yticks([t for t in ax.get_yticks() if 0 <= t <= 100])
    # Assigns the value on the y-axis to display the customer churn rate in percentage.
    ax.set_ylabel("Churn rate (%)")
    # Label for the x-axis and space between x-axis and its label are set here.
    ax.set_xlabel(xlabel, labelpad=10)
    # Apply the pre-defined chart style which includes the background color, title, borders, ticks and gridlines.
    style_axes(ax, title)
    # Chart legend is displayed in the top-left corner with 2 columns using the specified font size.
    ax.legend(loc="upper left", ncol=2, fontsize=13)

# Defines a function that helps in formatting the numerical value to a compact value.
def fmt_edge(v):
    """Compact number: 0 / 0.8 / 12 / 312 (no '-0.0', no trailing zeros)."""
    # Defines the way of formatting the numbers without using the negative zero or extra zero.
    # Checks for the value less than 0.05 and returns zero.
    if abs(v) < 0.05:
        return "0"
    # Defines the way of formatting the numbers having value >= 10 as integer without decimal.
    if abs(v) >= 10:
        return f"{v:.0f}"
    # Formats the values with one decimal point but without any extra zero and dot.
    return f"{v:.1f}".rstrip("0").rstrip(".")

# Defines a function to display a donut chart depicting the distribution of retained and churned customers.
def panel_donut(ax, df, counts, total, churn_rate):
    # Sets the background color of the chart area by using the defined card color.
    ax.set_facecolor(C_CARD)
    # Defines the chart title with appropriate padding, color and left alignment.
    ax.set_title("Customer Base Split", pad=16, color=C_DARK, loc="left")
    # Gets customer categories having count more than 0.
    present = counts.to_numpy() > 0
    # Generates the donut chart by using customer category count.
    # Assigns the predefined colors to retained and churned customer categories.
    wedges, _ = ax.pie(
        counts.to_numpy()[present],
        colors=np.array([C_RETAINED, C_CHURNED])[present], startangle=90,
        # Displays the customer categories in an anticlockwise manner.
        counterclock=False,
        # Defines the donut width, border color and border width.
        wedgeprops=dict(width=0.36, edgecolor="white", linewidth=4)
    )
    # Displays the customer churn rate as the center value of the donut chart.
    ax.text(0, 0.10, f"{churn_rate:.1f}%", ha="center", va="center",
            fontsize=36, fontweight="bold", color=C_CHURNED)
    # Defines the label of the churn rate below the center percentage value.
    ax.text(0, -0.18, "churn rate", ha="center", va="center",
            fontsize=17, fontweight="bold", color=C_BORDER)
    # Generates the legend labels with customer categories, count and percentage.
    names = [f"{o}: {int(c):,} ({100 * c / total:.1f}%)"
             for o, c, p in zip(ORDER, counts.to_numpy(), present) if p]
    # Displays the donut chart legend with customer count and percentages.
    ax.legend(wedges, names, loc="upper center", bbox_to_anchor=(0.5, 0.02),
              ncol=1, frameon=False, fontsize=16)

# Function for plotting churn rate of customers based on their subscription types
def panel_segment(ax, df, overall):
    # Subscription churn analysis title
    title = "Churn Rate by Subscription Type"
    # Check whether the column containing subscription type is not present or it has only missing values
    if "subscription_type" not in df or df["subscription_type"].isna().all():
        # Displays the message in the console in case there is no subscription type information available.
        no_data(ax, title, "subscription_type column not available")
        # Breaks out from function execution in case there is no subscription type information.
        return
    # Conversion of subscription type column values to string and replacing with "Missing" in case of missing values
    sub = df["subscription_type"].astype("string").fillna("Missing")
    # Grouping customers based on subscription types along with average churn rate and customer count.
    g = df.groupby(sub)["churn"].agg(["mean", "size"])
    # Sorting the subscription types in decreasing order of churn rate and taking top 8 subscription types
    g = g.sort_values("mean", ascending=False).head(8)
    # Creating churn rate bar plot for the selected subscription types
    # Converting the churn rate to percentage and giving customer counts and overall churn rate also
    rate_bars(ax, g.index.tolist(), g["mean"] * 100, g["size"], overall,
              # Using the BuPu color scale and rotating the labels by 25 degrees in case more than 4 categories are there
              title, "", cmap="BuPu", rotate=25 if len(g) > 4 else 0)

# Function to analyze customer churn rate depending on the recency of their activity
def panel_recency(ax, df, overall):
    # Choose the first available recency feature among the last login and last purchase columns that have more than 3 unique values
    col = next((c for c in ("days_since_last_login", "days_since_last_purchase")
                if has(df, c) and df[c].nunique() > 3), None)
    # Define the title of the recency-based churn analysis
    title = "Churn Rate by Recency"
    # Check if the recency feature can be used
    if col is None:
        # Show a placeholder chart in case there is no such recency feature
        no_data(ax, title)
        # Exit the function in case the recency feature is not present
        return
    # Select the recency feature and churn values, drop the NaNs and make a copy of the data
    d = df[[col, "churn"]].dropna().copy()
    # Create at most 6 quantiles from the recency feature values, fixing the possible duplicate bin edges
    d["bin"] = pd.qcut(d[col], q=min(6, d[col].nunique()), duplicates="drop")
    # Group the data by the recency intervals and calculate the average churn rate and customer number for each interval
    g = d.groupby("bin", observed=True).agg(rate=("churn", "mean"),
                                            n=("churn", "size"))
    # Calculate the min and max values of the selected recency feature among all available records
    lo_all, hi_all = d[col].min(), d[col].max()
    # Make an empty list for the formatted recency interval labels
    labels = []
    # Iterate through all the recency intervals and their indices
    for i, iv in enumerate(g.index):
        # Use the actual min recency value as the lower bound of the first recency interval
        lo = lo_all if i == 0 else iv.left
        # Uses the true maximum value of recency as the upper limit of the last interval.
        hi = hi_all if i == len(g) - 1 else iv.right
        # Transforms the interval boundaries into concise numeric codes.
        a_, b_ = fmt_edge(lo), fmt_edge(hi)
        # Appends one number if both of the interval boundaries are the same; otherwise, forms a range of numbers.
        labels.append(a_ if a_ == b_ else f"{a_}–{b_}")
    # Finds out the longest formatted interval label length.
    longest = max(len(t) for t in labels)
    # Builds a bar chart with recency intervals, their churn rates, number of customers and the total churn rate.
    rate_bars(ax, labels, g["rate"] * 100, g["n"], overall,
              # Uses the name of the chosen recency feature as the chart title and the name of the x-axis.

# Defines a function for comparing distribution of subscriptions' duration between churned and retained customers
def panel_tenure(ax, df):

    style_axes(ax, title)
# Defines a function for finding the best associations between customer features and churn visualization.
def panel_drivers(ax, df):
    # Defines the name of customer churn associations analysis.
    title = "Top Associations with Churn"
    # Initializes an empty list for saving feature names and their correlations.
    rows = []
    # Loops through all the features which are defined in LABELS dictionary.
    for col in LABELS:
        # Checks if the current feature exists and consists of valid values.
        if has(df, col):
            # Takes the current feature and churn values after dropping the records with missing values.
            p = df[[col, "churn"]].dropna()
            # Checks if the feature has at least 3 observations and not only 1 value and not only 1 churn value.
            if len(p) >= 3 and p[col].nunique() > 1 and p["churn"].nunique() > 1:
                # Finds Pearson correlation coefficient between the current feature and customer churn.
                r = p[col].corr(p["churn"])
                # Checks if the correlation value is a number (not infinity).
                if np.isfinite(r):
                    # Adds the feature label and the corresponding correlation value to results list.
                    rows.append((LABELS[col], r))
    # Checks if the feature associations exist.
    if not rows:
        # Plots the dummy chart if there are no valid associations.
        no_data(ax, title)
        # Interrupts the function execution if there are no valid associations.
        return
    # Orders the associations by absolute value of correlation in descending order and takes the first 8.
    rows = sorted(rows, key=lambda t: abs(t[1]), reverse=True)[:8][::-1]
    # Splits the feature names and correlation values from the ordered results.
    names, vals = zip(*rows)
    # Determines the maximal absolute correlation value.
    m_ = max(abs(v) for v in vals) or 1
    # Determines bar colours based on the direction and strength of correlation value.
    # Applies colour maps: Oranges for positive values and GnBu for negative values.
    colors = [(plt.cm.Oranges if v > 0 else plt.cm.GnBu)(0.35 + 0.6 * abs(v) / m_)
              for v in vals]
    # Plots horizontal bar plot based on feature names, correlation values and colours.
    # Sets parameters like bar border colour, bar border width, bar height and ordering.
    bars = ax.barh(names, vals, color=colors, edgecolor=C_BORDER,
                   linewidth=1.5, height=0.66, zorder=3)
    # Sets horizontal axis limit based on maximum correlation value adding additional plotting area.
    lim = max(0.5, m_ * 1.45)
    # Sets horizontal axis limit to be equal to at least 0.5.
    # Loops through each bar and its corresponding correlation value.
    for b, v in zip(bars, vals):
        # Labels each bar with correlation value including correct sign.
        ax.text(v + (lim * 0.03 if v >= 0 else -lim * 0.03),
                # Places positive correlation labels to the right of bars and negative to the left.
                b.get_y() + b.get_height() / 2, f"{v:+.2f}",
                # Places correlation labels vertically centred within corresponding bars.
                ha="left" if v >= 0 else "right", va="center",
                # Sets font size, bold and colour properties for correlation annotations.
                fontsize=15, fontweight="bold", color=C_DARK)

    # Draws vertical reference line at zero to distinguish between positive and negative correlation values.
    ax.axvline(0, color=C_BORDER, linewidth=2)
    # Sets symmetric horizontal axis limits to compare positive and negative associations.
    ax.set_xlim(-lim, lim)
    # Sets label for x-axis as Pearson correlation with customer churn.
    ax.set_xlabel("Pearson r with churn", labelpad=8)
    # Sets font size of y-axis ticks.
    ax.tick_params(axis="y", labelsize=16)
    # Applies the pre-defined chart formatting options and adds the grid lines on the x-axis.
    style_axes(ax, title, grid_axis="x")
    # Create customized legend items for providing an interpretation of positive and negative correlations.
    ax.legend(handles=[
        plt.Rectangle((0, 0), 1, 1, fc=plt.cm.Oranges(0.7), ec=C_BORDER,
                      label="Higher → more churn"),
        plt.Rectangle((0, 0), 1, 1, fc=plt.cm.GnBu(0.7), ec=C_BORDER,
                      label="Higher → less churn")],
        loc="lower right", fontsize=13)

# Function to plot a KPI card showing the metric, label and additional information about it
def kpi_card(ax, label, value, sub, color, bg):
    # Set the background color of the KPI card
    ax.set_facecolor(bg)
    # Hide the X-axis ticks of the chart
    ax.set_xticks([])
    # Hide the Y-axis ticks of the chart
    ax.set_yticks([])
    # Iterate over all axis border elements
    for s in ax.spines.values():
        # Hide all axis borders
        s.set_visible(False)
    # Draw a horizontal-colored line at the top of the KPI card
    ax.axhline(1, color=color, linewidth=10)
    # Set the limits for the horizontal axis to be from 0 to 1
    ax.set_xlim(0, 1)
    # Set the limits for the vertical axis to be from 0 to 1
    ax.set_ylim(0, 1)
    # Display the KPI value at the center of the card
    ax.text(0.5, 0.64, value, fontsize=44, fontweight="bold", color=color,
            ha="center", va="center", transform=ax.transAxes)

    # Display the label of the KPI in uppercase below the main value
    ax.text(0.5, 0.33, label.upper(), fontsize=18, fontweight="bold",
            color=C_DARK, ha="center", va="center", transform=ax.transAxes)
    # Display the additional information for the KPI label
    ax.text(0.5, 0.13, sub, fontsize=14, color="#546e7a",
            ha="center", va="center", transform=ax.transAxes)
# Defining a function that will help create textual insights using customer churn data
def make_insight(df, churn_rate):
    # Declaring an empty list to hold the discovered customer insights
    parts = []
    # Checking if the column with subscription type values is present in the dataset
    if "subscription_type" in df and df["subscription_type"].notna().any():
        # Grouping the customers by subscription type and computing the average churn rate
        r = df.groupby("subscription_type")["churn"].mean().sort_values()
        # Sorting the subscriptions in accordance to the average churn rate
        # Checking if there are multiple subscriptions to make a comparison between
        if len(r) > 1:
            # Identifying the subscription plan with the highest churn rate and computing the churn percentage
            parts.append(f"Highest-churn plan: {r.index[-1]} "
                         f"({100 * r.iloc[-1]:.1f}%)")
    # Declaring the best association as none before evaluating the available features
    best = None
    # Iterating through the customer features specified in the LABELS dictionary
    for col in LABELS:
        # Checking if the current feature is present in the dataframe and has valid values
        if has(df, col):
            # Selecting the current feature and churn values by dropping the null values
            p = df[[col, "churn"]].dropna()
            # Checking if there are at least 3 observations, at least 2 unique values of the feature and churn value
            if len(p) >= 3 and p[col].nunique() > 1 and p["churn"].nunique() > 1:
                # Computes the Pearson correlation coefficient between the selected feature and the customer churn.
                r = p[col].corr(p["churn"])
                # Determines if the correlation coefficient value is valid and if the coefficient is greater than the earlier found association.
                if np.isfinite(r) and (best is None or abs(r) > abs(best[1])):
                    # Records the label of the feature and its correlation coefficient when the feature represents the most significant association found till now.
                    best = (LABELS[col], r)
    # Determines if there is a valid most significant feature association found.
    if best:
        # Appends the most significant feature association and its correlation coefficient in the customer insights.
        parts.append(f"Strongest association: {best[0]} (r = {best[1]:+.2f})")
    # Joins all the insights found in one string, separated by vertical bars.
    # Returns the generated customer insights in a string format.
    return "   |   ".join(parts)

# Defines a function to construct the entire customer churn dashboard and save the generated image.
def build_dashboard(source, output_path="customer_churn_dashboard.png", show=True):
    # Takes the dataset source, output file path and show parameter as inputs.
    # Prepares and verifies the dataset using the prepared data preparation function.
    df = prepare_data(source)
    # Computes the total number of customer records in the dataset.
    total = len(df)
    # Computes the number of retained and churned customers and ensures both are accounted for.
    counts = df["churn"].value_counts().reindex([0, 1], fill_value=0)
    # Extracts the total number of retained and churned customers as integers.
    retained, churned = int(counts.loc[0]), int(counts.loc[1])
    # Computes the total customer churn rate as a percentage.
    churn_rate = 100 * churned / total
    # Verifies the availability and content of the subscription tenure column.
    if has(df, "subscription_tenure_days"):
        # Segregates the customer records into churned and retained groups and computes their average subscription tenures.
        t = df.groupby("churn")["subscription_tenure_days"].mean()
        # Computes the total average subscription tenure and formats it as a whole number and "day" unit.
        avg_tenure = f"{df['subscription_tenure_days'].mean():,.0f} d"
        # Constructs a support description that contrasts the average tenures of churned and retained customers.
        tenure_sub = (f"Churned {t.get(1, np.nan):.0f}d · Retained "
                      f"{t.get(0, np.nan):.0f}d")
        # Utilizes the average subscription tenure and support description if the subscription tenure data is present.
    else:
        # Returns N/A and a missing-column error message if no subscription tenure data is present.
        avg_tenure, tenure_sub = "N/A", "tenure column missing"
    # Constructs the main dashboard figure with the specified dimensions and background color.
    fig = plt.figure(figsize=(27, 18), facecolor=C_BG)
    # Sets the main dashboard title with the specified dimensions, bold formatting, color and vertical alignment.
    fig.suptitle("PerfectPetClub — Customer Churn Dashboard", fontsize=30,
                 fontweight="bold", color=C_DARK, y=0.985)
    # Extracts textual insights from the customer churn dataset using the defined insight function.
    insight = make_insight(df, churn_rate)
    # Checks if any customer insights have been created.
    if insight:
        # Displays the extracted insights on the center of the dashboard with the specified dimensions, color and bold formatting.
        fig.text(0.5, 0.947, insight, ha="center", fontsize=17,
                 color="#546e7a", fontweight="bold")
    # Constructs a separate grid for the KPI cards allowing independent modification of spacing.
    # Specifies a single row layout of five KPI cards with the defined figure borders and spacing.
    gs_k = gridspec.GridSpec(1, 5, figure=fig, left=0.06, right=0.98,
                             top=0.915, bottom=0.80, wspace=0.12)
    # Defines the KPI collection with the metric names, calculated values, descriptions and colors.
    kpis = [("Total Customers", f"{total:,}", "customer records", C_VIOLET, "#ede7f6"),
            # Defines the total customers KPI using the total number of customer records.
            ("Churned", f"{churned:,}", f"{churn_rate:.1f}% of base", C_CHURNED, "#fbe9e7"),
            # Defines the churned customers KPI using the number of churned customers and churn percentage.
            ("Retained", f"{retained:,}", f"{100 * retained / total:.1f}% of base", C_RETAINED, "#e0f2f1"),
            # Defines the retained customers KPI using the number of retained customers and retention percentage.
            ("Churn Rate", f"{churn_rate:.1f}%",
             f"≈ 1 in {total / churned:.1f} customers" if churned else "no churn",
             "#bf360c", "#ffebee"),
            # Defines the churn rate KPI and the approximate number of customers corresponding to one churned customer.
            # Displays a no-churn message in case there are no churned customers.
            ("Avg Tenure", avg_tenure, tenure_sub, C_BLUE, "#e3f2fd")]
            # Defines the average tenure KPI using the calculated average tenure value and its description.
    # Loop over all five combinations of the KPI and the position it is supposed to have.
    for i, (lab, val, sub, col, bg) in enumerate(kpis):
        # Create one subplot per each KPI card and show it with the KPI card plotting function.
        kpi_card(fig.add_subplot(gs_k[0, i]), lab, val, sub, col, bg)
    # Create a chart grid of the plot that has two rows and three columns.
    gs = gridspec.GridSpec(2, 3, figure=fig, left=0.075, right=0.98,
                           # Set the figure boundaries, the vertical space, and the horizontal space for the six dashboard plots.
                           top=0.725, bottom=0.075, hspace=0.50, wspace=0.42)
    # Create an axes object for each position of the two-row and three-column chart.
    ax = [fig.add_subplot(gs[r, c]) for r in range(2) for c in range(3)]
    # Plot the distribution of the retained and churned customers by means of a donut chart.
    panel_donut(ax[0], df, counts, total, churn_rate)
    # Plot customer churn rates by subscription type.
    panel_segment(ax[1], df, churn_rate)
    # Plot customer churn rates by customer activity recency.
    panel_recency(ax[2], df, churn_rate)
    # Compare the subscription tenure distributions of retained and churned customers.
    panel_tenure(ax[3], df)
    # Compare the total payment amount distributions of retained and churned customers.
    panel_payment(ax[4], df)
    # Plot the most significant relations between customer attributes and churn.
    panel_drivers(ax[5], df)
    # Converts the given path for the dashboard output to a Path and expands the user path if needed.
    output_path = Path(output_path).expanduser()
    # Creates the parent directory for the dashboard output file, creating any necessary intermediate directories.
    output_path.parent.mkdir(parents=True, exist_ok=True)
    # Writes the completed dashboard as a PNG file with the given resolution and tight bounding box and background colour.
    fig.savefig(output_path, dpi=100, bbox_inches="tight", facecolor=C_BG)
    # Prints out the confirmation message that includes the absolute path to the saved dashboard.
    print(f"✓ Dashboard saved: {output_path.resolve()}")
    # Determines whether the dashboard should be displayed.
    if show:
        # Shows the dashboard if the show flag is set to True.
        plt.show()
    else:
        # Closes the dashboard figure if the show flag is set to False.
        plt.close(fig)
    # Returns the output path to the saved dashboard file.
    return output_path

# Defines the primary method that runs the customer churn dashboard application.
def main(argv=None):
    # Reads the command-line arguments passed in the argv parameter.
    # Gets the base path where the input and output files are located.
    # Uses the /content path in case of its existence; otherwise, the current working path is used.
    base = Path("/content") if Path("/content").is_dir() else Path.cwd()
    # Defines an argument parser with the description of the customer churn dashboard.
    p = argparse.ArgumentParser(description="Customer churn dashboard.")
    # Defines the input argument with the path to the dataset file, using the default Parquet file path.
    p.add_argument("--input", type=Path, default=base / "customer_churn.parquet")
    # Defines the output argument with the path to the dashboard image file, using the default PNG file path.
    p.add_argument("--output", type=Path, default=base / "customer_churn_dashboard.png")
    # Defines the no-show argument that disables displaying the dashboard in case of its usage.
    p.add_argument("--no-show", action="store_true")
    # Parses the command-line arguments, saving the provided ones.
    a = p.parse_args(argv)
    # Verifies the existence of the specified input dataset file.
    if not a.input.is_file():
        # Throws a FileNotFoundError exception with the dataset path in case the file is not found.
        raise FileNotFoundError(f"Dataset not found: {a.input}")
    # Suppresses FutureWarning exceptions to exclude any extra warning in case of dashboard execution.
    warnings.filterwarnings("ignore", category=FutureWarning)
    # Reads the Parquet dataset and creates the dashboard with the specified output path.
    return build_dashboard(pd.read_parquet(a.input), a.output, show=not a.no_show)
